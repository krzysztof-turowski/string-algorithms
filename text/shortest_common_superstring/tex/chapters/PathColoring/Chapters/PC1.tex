\section{The Path-Coloring Lemma} \label{sec:path_coloring}
\begin{lemma}[Path-Coloring Lemma \cite{Blaser2004}]
Let $G$ be a directed graph satisfying the following structural properties:
\begin{enumerate}
    \item Every vertex has an indegree of at most 2.
    \item Every vertex has an outdegree of at most 2.
    \item The total degree of any vertex (indegree plus outdegree) is at most 3.
    \item The graph $G$ contains no 2-cycles.
\end{enumerate}
Then the edges of $G$ can be colored using exactly two colors such that the
edges in each color class form a collection of directed paths.
\end{lemma}
